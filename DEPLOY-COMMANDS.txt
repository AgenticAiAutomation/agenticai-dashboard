PHASE 2: VPS DATABASE SETUP
=============================================================================

# SSH to VPS
ssh your-user@your-vps-ip

# Create database
sudo -u postgres psql <<EOF
CREATE DATABASE agenticai_dashboard;
CREATE USER dashboard_user WITH PASSWORD 'YourSecurePassword123';
GRANT ALL PRIVILEGES ON DATABASE agenticai_dashboard TO dashboard_user;
ALTER DATABASE agenticai_dashboard OWNER TO dashboard_user;
\q
EOF

# Test connection (should prompt for password, then show psql prompt)
psql -U dashboard_user -d agenticai_dashboard -h localhost
# Type \q to exit

# If connection test succeeds, database is ready ✓


=============================================================================
PHASE 3: CLONE CODE TO VPS
=============================================================================

# Still on VPS, navigate to web root
cd /var/www

# Clone repository (replace with your GitHub URL)
sudo git clone https://github.com/YOUR_USERNAME/agenticai-dashboard.git

# Set permissions
sudo chown -R $USER:$USER /var/www/agenticai-dashboard
cd /var/www/agenticai-dashboard

# Verify files
ls -la
# You should see: api/, web/, infra/, scripts/, README.md, etc.


=============================================================================
PHASE 4: CONFIGURE BACKEND
=============================================================================

cd /var/www/agenticai-dashboard/api

# Copy .env template
cp .env.example .env

# Edit .env file
nano .env

# UPDATE THESE VALUES:
# - DATABASE_URL password (replace 'YourSecurePassword123')
# - JWT_SECRET_KEY (generate with: openssl rand -hex 32)
# - INITIAL_OWNER_PASSWORD (choose secure password)
# - CORS_ORIGINS (should be: https://dashboard.agenticaiautomation.co,http://localhost:3000)

# Save: Ctrl+O, Enter, Ctrl+X

# Create virtual environment
python3 -m venv venv

# Activate venv
source venv/bin/activate

# Install dependencies
pip install -r requirements.txt

# This takes 2-3 minutes, be patient

PHASE 5: RUN MIGRATIONS & SEED DATABASE
=============================================================================

PHASE 7: CONFIGURE NGINX
=============================================================================

# Copy nginx config
sudo cp /var/www/agenticai-dashboard/infra/nginx-dashboard.conf /etc/nginx/sites-available/dashboard

# IMPORTANT: Before enabling, comment out SSL lines since cert doesn't exist yet
sudo nano /etc/nginx/sites-available/dashboard

# Comment out these lines (add # at start):
# ssl_certificate /etc/letsencrypt/...
# ssl_certificate_key /etc/letsencrypt/...
# include /etc/letsencrypt/...
# ssl_dhparam /etc/letsencrypt/...

# Keep the "listen 443 ssl http2;" line but rest can be commented

# Save and exit

# Create symlink
sudo ln -sf /etc/nginx/sites-available/dashboard /etc/nginx/sites-enabled/

# Test nginx config
sudo nginx -t

# Should say: "syntax is ok" and "test is successful"

# Reload nginx
sudo systemctl reload nginx
